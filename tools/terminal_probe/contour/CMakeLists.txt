cmake_minimum_required(VERSION 3.25)
project(lapis_contour_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(LAPIS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${LAPIS_SOURCE_DIR}/cmake/ProjectOptions.cmake")
# Instrument the engine and dependencies as well as the first-party consumer.
if(lapis_sanitizer_flags)
    add_compile_options(${lapis_sanitizer_flags})
    add_link_options(${lapis_sanitizer_flags})
endif()
set(CONTOUR_SOURCE_DIR "" CACHE PATH "Pinned Contour source directory")
set(CONTOUR_DEPS_DIR "" CACHE PATH "Pinned dependency source directory")
if(NOT EXISTS "${CONTOUR_SOURCE_DIR}/src/vtbackend/CMakeLists.txt")
    message(FATAL_ERROR "Provide CONTOUR_SOURCE_DIR from the pinned experiment sources")
endif()
find_package(Threads REQUIRED)
set(GSL_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_TIDY OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_TESTING OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_BENCHMARK OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_TOOLS OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BOXED_TESTING OFF CACHE BOOL "" FORCE)
set(REFLECTION_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${CONTOUR_DEPS_DIR}/gsl" deps/gsl EXCLUDE_FROM_ALL)
add_subdirectory("${CONTOUR_DEPS_DIR}/unicode" deps/unicode EXCLUDE_FROM_ALL)
add_subdirectory("${CONTOUR_DEPS_DIR}/boxed" deps/boxed EXCLUDE_FROM_ALL)
add_subdirectory("${CONTOUR_DEPS_DIR}/reflection" deps/reflection EXCLUDE_FROM_ALL)

# Upstream's library CMake files use PROJECT_SOURCE_DIR for public include paths.
# This function confines that upstream convention to the headless library graph.
function(add_contour_engine)
    set(PROJECT_SOURCE_DIR "${CONTOUR_SOURCE_DIR}")
    set(PROJECT_VERSION_MAJOR 0)
    set(PROJECT_VERSION_MINOR 0)
    set(PROJECT_VERSION_PATCH 0)
    set(CONTOUR_VERSION_STRING "6777ff05014f8ff163b071e8b0e942830119db80")
    set(CONTOUR_TESTING OFF)
    set(CONTOUR_WITH_UTEMPTER OFF)
    set(CONTOUR_TRACY OFF)
    set(CONTOUR_TRACY_TOOLS OFF)
    set(CONTOUR_TRACY_GUI OFF)
    include("${CONTOUR_SOURCE_DIR}/cmake/Tracy.cmake")
    add_subdirectory("${CONTOUR_SOURCE_DIR}/src/crispy" upstream/crispy EXCLUDE_FROM_ALL)
    add_subdirectory("${CONTOUR_SOURCE_DIR}/src/vtparser" upstream/vtparser EXCLUDE_FROM_ALL)
    add_subdirectory("${CONTOUR_SOURCE_DIR}/src/vtpty" upstream/vtpty EXCLUDE_FROM_ALL)
    add_subdirectory("${CONTOUR_SOURCE_DIR}/src/vtbackend" upstream/vtbackend EXCLUDE_FROM_ALL)
endfunction()
add_contour_engine()

add_executable(lapis_contour_probe engine.cpp "${LAPIS_SOURCE_DIR}/tools/terminal_probe/replay.cpp")
target_link_libraries(lapis_contour_probe PRIVATE lapis_project_options vtbackend)
# Upstream headers are external code; keep lapis warnings on the adapter itself.
target_include_directories(lapis_contour_probe SYSTEM PRIVATE
    "${CONTOUR_SOURCE_DIR}/src"
    "${CONTOUR_DEPS_DIR}/gsl/include"
    "${CONTOUR_DEPS_DIR}/unicode/src"
    "${CONTOUR_DEPS_DIR}/boxed/include"
    "${CONTOUR_DEPS_DIR}/reflection/include")
enable_testing()
add_test(NAME contour-terminal-replay COMMAND lapis_contour_probe)
set_tests_properties(contour-terminal-replay PROPERTIES TIMEOUT 30)
